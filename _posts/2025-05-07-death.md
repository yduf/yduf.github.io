// Split-flap day counter: starts at 10,410 on 18/09/2026 and decreases by 1 each day.
// Uses UTC calendar dates, so the counter drops by one at midnight UTC.
(function () {
  var START_COUNT = 10410;
  var START_DATE = new Date(Date.UTC(2026, 8, 18)); // 18 Sept 2026 (month is 0-based)
  var DAY_MS = 24 * 60 * 60 * 1000;

  var now = new Date();
  var today = Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate());
  var daysElapsed = Math.floor((today - START_DATE.getTime()) / DAY_MS);
  var daysLeft = START_COUNT - daysElapsed;

  var container = document.getElementById('days-left-counter');
  var text = daysLeft.toLocaleString('en-US');
  container.setAttribute('aria-label', text + ' days left');

  // Build one flip card per character (comma is a plain separator)
  var cards = [];
  for (var i = 0; i < text.length; i++) {
    var ch = text[i];
    if (ch === ',') {
      var sep = document.createElement('span');
      sep.className = 'sep';
      sep.textContent = ',';
      container.appendChild(sep);
      continue;
    }
    var flap = document.createElement('span');
    flap.className = 'flap';
    ['top', 'bottom'].forEach(function (pos) {
      var half = document.createElement('span');
      half.className = 'half ' + pos;
      var digit = document.createElement('span');
      digit.textContent = ch;
      half.appendChild(digit);
      flap.appendChild(half);
    });
    container.appendChild(flap);
    cards.push(flap);
  }

  function setDigit(flap, digit) {
    flap.firstChild.firstChild.textContent = digit;
    flap.lastChild.firstChild.textContent = digit;
  }

  // Animate like a split-flap board: digits cycle, then settle (staggered)
  var reducedMotion = window.matchMedia &&
    window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  if (reducedMotion) {
    cards.forEach(function (flap, i) { setDigit(flap, text.replace(',', '')[i] || text[i]); });
  } else {
    cards.forEach(function (flap, i) {
      var target = text.replace(',', '')[i];
      var cycles = 8 + i * 3, n = 0;
      var iv = setInterval(function () {
        n++;
        if (n >= cycles) {
          setDigit(flap, target);
          clearInterval(iv);
        } else {
          setDigit(flap, String(Math.floor(Math.random() * 10)));
        }
      }, 70);
    });
  }
})();
</script>
