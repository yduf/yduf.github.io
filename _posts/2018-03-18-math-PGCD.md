# [pseudo code ⮺](https://en.wikipedia.org/wiki/Euclidean_algorithm)

{% highlight cpp %}
function gcd(a, b)
    while b ≠ 0
       t := b; 
       b := a mod b; 
       a := t; 
    return a;
{% endhighlight %}

or

{% highlight cpp %}
function gcd(a, b)
    if b = 0
       return a; 
    else
       return gcd(b, a mod b);
{% endhighlight %}
